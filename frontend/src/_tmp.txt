import { useMemo, useRef } from "react"

import * as THREE from "three"

type Node = { position: [number, number, number]; color: string }

const PALETTE = ["#22D3EE", "#3B82F6", "#8B5CF6", "#60A5FA"]

function Network({ count = 32 }: { count?: number }) {
  const group = useRef<THREE.Group>(null)

  const nodes: Node[] = useMemo(() => {
    const out: Node[] = []
    for (let i = 0; i < count; i++) {
      const r = 2.2 + Math.random() * 1.4
      const theta = Math.random() * Math.PI * 2
      const phi = Math.acos(2 * Math.random() - 1)
      out.push({
        position: [
          r * Math.sin(phi) * Math.cos(theta),
          r * Math.cos(phi),
          r * Math.sin(phi) * Math.sin(theta),
        ],
        color: PALETTE[i % PALETTE.length],
      })
    }
    return out
  }, [count])

  const edges = useMemo(() => {
    const pairs: Array<[[number, number, number], [number, number, number]]> = []
    for (let i = 0; i < nodes.length; i++) {
      for (let j = i + 1; j < nodes.length; j++) {
        const a = new THREE.Vector3(...nodes[i].position)
        const b = new THREE.Vector3(...nodes[j].position)
        if (a.distanceTo(b) < 2.6) pairs.push([nodes[i].position, nodes[j].position])
      }
    }
    return pairs.slice(0, 100)
  }, [nodes])

  useFrame((state, delta) => {
    if (!group.current) return
    group.current.rotation.y += delta * 0.08
    group.current.rotation.x = Math.sin(state.clock.elapsedTime * 0.15) * 0.15
  })

  return (
    <group ref={group}>
      {nodes.map((n, i) => (
        <mesh key={i} position={n.position}>
          <sphereGeometry args={[0.07, 12, 12]} />
          <meshStandardMaterial color={n.color} emissive={n.color} emissiveIntensity={1.4} />
        </mesh>
      ))}
      {edges.map((e, i) => (
        <Line
          key={i}
          points={[e[0], e[1]]}
          color="#3B82F6"
          lineWidth={0.7}
          transparent
          opacity={0.28}
        />
      ))}
    </group>
  )
}

export function DataNetwork() {
  return (
    <div style={{ width: "100%", height: "100%", minHeight: 480 }}>
      <Canvas
        dpr={[1, 2]}
        camera={{ position: [0, 0, 7], fov: 50 }}
        gl={{ antialias: true, alpha: true }}

        <pointLight position={[5, 5, 5]} intensity={1.5} color="#60A5FA" />

}

import { Component, type ErrorInfo, type ReactNode } from "react"
import { DataNetwork } from "./DataNetwork"

type Props = { children?: ReactNode }
type State = { crashed: boolean }

class ErrorBoundary extends Component<Props, State> {
  state: State = { crashed: false }

  static getDerivedStateFromError(): State {
    return { crashed: true }
  }

  componentDidCatch(error: Error, info: ErrorInfo) {
    console.warn("[Scene3D] disabled due to error:", error.message)
    console.warn(info.componentStack)
  }

  render() {
    if (this.state.crashed) {
      return <GradientFallback />
    }
    return this.props.children
  }
}

function GradientFallback() {
  return (
    <div className="w-full h-full min-h-[480px] relative overflow-hidden">
      <div className="absolute inset-0 bg-[radial-gradient(circle_at_30%_20%,rgba(34,211,238,0.18),transparent_55%),radial-gradient(circle_at_70%_80%,rgba(59,130,246,0.20),transparent_55%)]" />
      <svg className="absolute inset-0 w-full h-full opacity-30" xmlns="http://www.w3.org/2000/svg">
        <defs>
          <pattern id="grid" width="48" height="48" patternUnits="userSpaceOnUse">
            <path d="M 48 0 L 0 0 0 48" fill="none" stroke="rgba(34,211,238,0.15)" strokeWidth="0.5" />
          </pattern>
        </defs>
        <rect width="100%" height="100%" fill="url(#grid)" />
      </svg>
    </div>
  )
}

import { createContext, useCallback, useContext, useMemo, useState } from "react"
import { login as apiLogin, register as apiRegister, type RegisterPayload } from "@/api/auth"
import { tokens } from "@/lib/tokens"

type AuthCtx = {
  isAuthenticated: boolean
  login: (u: string, p: string) => Promise<void>
  signup: (payload: RegisterPayload) => Promise<void>
  logout: () => void
}

const Ctx = createContext<AuthCtx | null>(null)

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [isAuthenticated, setAuth] = useState(tokens.has())

  const login = useCallback(async (u: string, p: string) => {
    const { access, refresh } = await apiLogin(u, p)
    tokens.set(access, refresh)
    setAuth(true)
  }, [])

  const signup = useCallback(async (payload: RegisterPayload) => {
    await apiRegister(payload)
    const { access, refresh } = await apiLogin(payload.username, payload.password)
    tokens.set(access, refresh)
    setAuth(true)
  }, [])

  const logout = useCallback(() => {
    tokens.clear()
    setAuth(false)
  }, [])

  const value = useMemo(
    () => ({ isAuthenticated, login, signup, logout }),
    [isAuthenticated, login, signup, logout],
  )
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>
}

export function useAuth() {
  const ctx = useContext(Ctx)
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>")
  return ctx
}

const RULES = [
  { id: "len", label: "At least 8 characters", test: (v: string) => v.length >= 8 },
  { id: "num", label: "Contains a number", test: (v: string) => /\d/.test(v) },
  { id: "alpha", label: "Contains a letter", test: (v: string) => /[A-Za-z]/.test(v) },
  { id: "special", label: "Contains a symbol (!@#$…)", test: (v: string) => /[^A-Za-z0-9]/.test(v) },
]

export function SignupPage() {
  const { signup } = useAuth()
  const nav = useNavigate()
  const [username, setUsername] = useState("")
  const [email, setEmail] = useState("")
  const [password, setPassword] = useState("")
  const [password2, setPassword2] = useState("")
  const [error, setError] = useState<string | null>(null)
  const [busy, setBusy] = useState(false)

  const rules = useMemo(() => RULES.map((r) => ({ ...r, ok: r.test(password) })), [password])
  const match = password.length > 0 && password === password2
  const canSubmit = username.trim().length > 0 && email.includes("@") && rules.every((r) => r.ok) && match

  async function onSubmit(e: FormEvent) {
    e.preventDefault()
    if (!canSubmit) { setError("Please satisfy all requirements."); return }
    setBusy(true); setError(null)
    try {
      await signup({ username: username.trim(), email: email.trim(), password, password2 })
      nav("/", { replace: true })
    } catch (err: any) {
      const data = err?.response?.data
      if (data && typeof data === "object") {
        const first = Object.values(data)[0]
        setError(typeof first === "string" ? first : "Signup failed.")
      } else setError("Signup failed. Please try again.")
    } finally {
      setBusy(false)
    }
  }

  return (
    <div className="min-h-screen grid lg:grid-cols-2 bg-[#0B1220] text-slate-100">
      <div className="flex items-center justify-center p-6 bg-[#0E1628] order-2 lg:order-1">
        <form onSubmit={onSubmit} className="w-full max-w-sm space-y-5 glass rounded-2xl p-8 shadow-2xl">
          <div>
            <h2 className="text-2xl font-semibold">Create your account</h2>
            <p className="text-sm text-slate-300 mt-1">Get started with WCIP</p>
          </div>
          <Input label="Username" value={username} onChange={(e) => setUsername(e.target.value)} required autoFocus />
          <Input label="Email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} required />
          <div>
            <Input label="Password" type="password" value={password} onChange={(e) => setPassword(e.target.value)} required />
            <ul className="mt-2 space-y-1">
              {rules.map((r) => (
                <li key={r.id} className={`text-xs ${r.ok ? "text-emerald-400" : "text-slate-400"}`}>
                  {r.ok ? "✓" : "○"} {r.label}
                </li>
              ))}
            </ul>
          </div>
          <Input label="Confirm password" type="password" value={password2} onChange={(e) => setPassword2(e.target.value)} error={password2 && !match ? "Passwords do not match" : undefined} required />
          {error && <div className="text-sm text-red-400 bg-red-400/10 border border-red-400/20 rounded-lg px-3 py-2">{error}</div>}
          <Button type="submit" block disabled={busy || !canSubmit}>{busy ? "Creating…" : "Create account"}</Button>
          <p className="text-xs text-center text-slate-300">
            Already have an account? <Link to="/login" className="text-cyan-400 hover:text-cyan-300">Sign in</Link>
          </p>
        </form>
      </div>
      <div className="hidden lg:block relative overflow-hidden order-1 lg:order-2">
        <div className="absolute inset-0"><Scene3D /></div>
        <div className="relative z-10 h-full flex flex-col items-center justify-center p-10 text-center">
          <div className="text-4xl font-semibold"><span className="text-cyan-400">W</span>CIP</div>
          <p className="mt-3 text-slate-300 text-sm max-w-xs">
            Join your organization's workforce intelligence platform.
          </p>
        </div>
      </div>
    </div>
  )
}

import ReactDOM from "react-dom/client"
import { QueryClient, QueryClientProvider } from "@tanstack/react-query"
import { RouterProvider } from "react-router-dom"
import { AuthProvider } from "@/features/auth/AuthContext"
import { router } from "@/router"
import "./index.css"

const qc = new QueryClient({
  defaultOptions: { queries: { retry: 1, refetchOnWindowFocus: false } },
})

ReactDOM.createRoot(document.getElementById("root")!).render(
  <React.StrictMode>
    <QueryClientProvider client={qc}>
      <AuthProvider>
        <RouterProvider router={router} />
      </AuthProvider>
    </QueryClientProvider>
  </React.StrictMode>,
)
